% ---------------------------------------------------------------------------
% Draft flags: material drafted mechanically and awaiting the author's check.
%   \problemdraft[note]{statement}{solution}
%       an exercise not yet verified. To ACCEPT it, delete "draft" from the
%       command name and drop any [note] -- a one-word edit.
%   \unverified{...}
%       inline marker for one specific claim inside otherwise checked material.
% \draftflagsfalse typesets a clean copy; tools/list_draft_flags.py reports
% everything still outstanding, and `make lint` counts them.
\newif\ifdraftflags \draftflagstrue
\definecolor{draftflagcolour}{RGB}{176,72,0}
\newcommand{\draftflagtag}[1]{%
  \ifdraftflags
    {\footnotesize\sffamily\color{draftflagcolour}%
      [\,unverified draft --- author check pending%
       \if\relax\detokenize{#1}\relax\else: #1\fi\,]}%
    \par\nobreak
  \fi}
\newcommand{\problemdraft}[3][]{\problem{\draftflagtag{#1}#2}{#3}}
\newcommand{\unverified}[1]{%
  \ifdraftflags{\color{draftflagcolour}#1\,{\footnotesize\sffamily[check]}}\else#1\fi}

% Identity matrix.  NOTE: \mathbb{1} is NOT valid with amssymb -- the msbm font
% has no digits, so \mathbb{1} silently typesets as the logic symbol U+22AE.
% Use a proper blackboard-bold one where the font is available, and fall back to
% an upright bold 1 otherwise, so the manuscript is portable.
\IfFileExists{bbm.sty}{%
  \usepackage{bbm}\newcommand{\Id}{\mathbbm{1}}}{%
  \IfFileExists{dsfont.sty}{%
    \usepackage{dsfont}\newcommand{\Id}{\mathds{1}}}{%
    \newcommand{\Id}{\mathbf{1}}}}
