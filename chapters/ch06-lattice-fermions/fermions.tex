\section{Fermions in QFT}

\begin{itemize}
\item  Here we work in $4D$ unless otherwise noted
  \item In the path integral\aidx{path integral} approach, fermions enter as Grassmann-valued anticommuting objects
\end{itemize}

\begin{equation}
\psi_{\alpha}(x) \psi_{\beta}(y)=-\psi_{\beta}(y) \psi_{\alpha}(x)
\end{equation}

\begin{itemize}
  \item In the continuum, the action for a single free Dirac fermion (we will primarily use Dirac fermions) is
\end{itemize}

\begin{equation}
S_{F}=\int d x\left[\bar{\psi}(x) \gamma_{\mu} \partial_{\mu} \psi(x)+m \bar{\psi}(x) \psi(x)\right]=\int d x d y \bar{\psi}(y) M(x, y) \psi(y)
\end{equation}

where \(\psi, \bar{\psi}\) are independent degrees of freedom and the Dirac matrices\aidx{Dirac matrices, Euclidean} in Euclidean metric satisfy

\begin{equation}
\left\{\gamma_{\mu,} \gamma_{\nu}\right\}=2 g_{\mu \nu} \mathbb{I}, \quad \gamma_{\mu}^{\dagger}=\gamma_{\mu}
\end{equation}

where the relation to Minkowski matrices is

\begin{equation}
\gamma_{4}=\gamma_{0}^{M} \quad \gamma_{j}=-i \gamma_{j}^{M}
\end{equation}

The matrix $\gamma_5$ is

\begin{equation}
\gamma_{5}=\gamma_{1} \gamma_{2} \gamma_{3} \gamma_{4} \quad \gamma_{5}^{2}=\mathbb{I} \quad \gamma_{5}^{\dagger}=\gamma_{5}
\end{equation}

\begin{itemize}
  \item The partition function\aidx{partition function} with external Grassmann sources\aidx{Grassmann variables} is
\end{itemize}

\begin{equation}
Z(\eta, \bar{\eta})=\int \mathcal{D} \bar{\psi} \mathcal{D} \psi \exp \left[-S-\int d x\{\bar{\psi}(x) \eta(x)+\bar{\eta}(x) \psi(x)\}\right]
\end{equation}

and correlation functions\aidx{correlation function}

\begin{equation}
    \langle \psi(x_1) \ldots \psi(x_j) \bar{\psi}(y_1) \ldots \bar{\psi}(y_k)\rangle = \frac{1}{Z[0,0]}\frac{\delta}{\delta \bar{\eta}(x_1)} \ldots \frac{\delta}{\delta \bar{\eta}(x_j)}\frac{\delta}{\delta \eta(y_1)} \ldots \frac{\delta}{\delta \eta(y_k)}Z[\eta, \bar{\eta}]|_{\eta, \bar{\eta} = 0}.
\end{equation}

\begin{itemize}
  \item Since the action is bilinear in $\bar{\psi} \psi$, the integrations can be performed exactly
\end{itemize}

\begin{align}
    Z[\eta, \bar{\eta}] &= \int \mathcal{D}\bar{\psi}\mathcal{D}\psi \exp[-
    \overset{\int\text{d}x \text{d}y \bar{\psi}M(x,y)\psi(y)}{\overbrace{\bar{\psi} \cdot M \cdot \psi} - \bar{\eta}\cdot \psi - \bar{\psi} \cdot \eta]} \nonumber\\
    &= \int \mathcal{D}\bar{\psi}^\prime \mathcal{D}\psi^\prime \exp[-\bar{\psi}^\prime \cdot M \cdot \psi^\prime + \bar{\eta}\cdot M^{-1} \cdot \eta] \nonumber \\
    &= \operatorname{det}[M]\exp[-\bar{\eta}\cdot M^{-1}\cdot \eta],
\end{align}
where we have made use of the change of variables $\psi^\prime = \psi + M^{-1}\eta, \bar{\psi}^\prime = \bar{\psi} + \bar{\eta} M^{-1}$. Then, noting
\begin{equation}
    \frac{\delta}{\delta{\bar{\eta}}(x)} \frac{\delta}{\delta \eta(y)} \exp[- \bar{\eta}\cdot M^{-1}\cdot \eta] = M^{-1}(x,y)\exp[- \bar{\eta}\cdot M^{-1}\cdot \eta],
\end{equation}
we find that
\begin{equation}
    \langle \psi(x_1)\ldots\psi(x_j)\bar{\psi}(y_1)\ldots\bar{\psi}(y_j)\rangle = (-1)^j \sum_{\text{perms } P}\operatorname{sign}(P)M^{-1}(x_1, y_{P(1)})\ldots M^{-1}(x_j, y_{P(j)})
\end{equation}

\begin{itemize}
  \item If there are \(N_{f}\) flavours of fermions transforming in the fundamental representation of a global symmetry group $G$
\end{itemize}
\begin{equation}
    \psi = \begin{pmatrix}
        \psi_1 \\ \vdots \\ \psi_{N_f}
    \end{pmatrix} \rightarrow \psi^\prime = \Omega(\theta)\psi, \quad \Omega(\theta) = e^{i \theta_a T_a},
\end{equation}
where the $T_a$ are the generators of $G$, then the action

\begin{equation}
S=\sum_{f=1}^{N_f} \int {\text{d} x} \bar{\psi}_{f}(x)\left(\gamma_{\mu} \partial_{\mu}+M_{f}\right) \psi_{f}(x)
\end{equation}

is invariant if \(M_{f}=M \quad \forall f\)

\begin{itemize}
  \item Promoting \(\partial_{\mu} \rightarrow D_{\mu}=\partial_{\mu}-i g A_{\mu}\) (the gauge covariant derivative), the action
\end{itemize}

\begin{equation}
S=\sum_{f=1}^{N_{f}} \int d x \bar{\psi}_{f, i}(x)\left(\gamma_\mu D_{\mu, i j}+M_{f} \delta_{i j}\right) \psi_{f, j}(x)
\end{equation}

is invariant under local gauge transformations under group $\tilde{G}$ where
\begin{align}
    \psi(x) \rightarrow \psi^\prime(x) =& \Omega(\theta(x)) \psi(x) \quad \bar{\psi}(x) \rightarrow \bar{\psi}^\prime(x) = \bar{\psi}(x)\Omega^{-1}(\theta(x)) \nonumber \\
    &D_\mu \rightarrow D_\mu^\prime = \Omega(\theta(x)) D_\mu \Omega(\theta(x))^\dagger
\end{align}
for
\begin{equation}
    \Omega(\theta(x)) = e^{i \theta_a(x) \lambda_a}, \quad \lambda_a \in D_f(a) \text{ are generators of }\tilde{G}
\end{equation}

\begin{itemize}
  \item For massless fermions, the action has various global symmetries
  \item Global vector rotations as above (also a symmetry if \(M_{f}=M\) )
\end{itemize}

\begin{equation}
\begin{aligned}
U_V(1) : \qquad & \psi \rightarrow e^{-\frac{i}{2} \alpha} \psi, \qquad \bar{\psi} \rightarrow \bar{\psi} e^{\frac{i}{2} \alpha}  \\
G\text{ (e.g. }SU(N_f)): \qquad & \psi \rightarrow e^{-\frac{i}{2} \alpha_{a} T_{a}}\psi, \qquad \bar{\psi} \rightarrow \bar{\psi} e^{\frac{i}{2} \alpha_{a} T_{a}}  
\end{aligned}
\end{equation}
\begin{itemize}
    \item Global axial rotations\aidx{UA1@$U_A(1)$ symmetry}
\end{itemize}
\begin{equation}
\begin{aligned}
U_{A}(1) : \qquad & \psi \rightarrow \exp \left(-\frac{i}{2} \alpha \gamma_{5}\right) \psi,\qquad  \bar{\psi} \rightarrow \bar{\psi} \exp \left(\frac{i}{2} \alpha \gamma_{5}\right) \\
G : \qquad& \psi \rightarrow \exp \left(-\frac{i}{2} \alpha_{a} T_{a} \gamma_{5}\right) \psi, \bar{\psi} \rightarrow \bar{\psi} \exp \left(\frac{i}{2} \alpha_{a} T_{a} \gamma_{5}\right) 
\end{aligned}
\end{equation}

\begin{itemize}
  \item Corresponding conserved (Noether) currents
\end{itemize}
\begin{equation}
V^{\mu a}=\frac{1}{2} \bar{\psi} \gamma_{\mu} T_{a} \psi, \qquad A^{\mu a}=\frac{1}{2} \bar{\psi} \gamma_{\mu} \gamma_{5} T_{a} \psi, \qquad \begin{gathered}
a=0,1, \ldots . \\
T_{0}=1
\end{gathered}
\end{equation}

\begin{itemize}
  \item Combining these two sets of symmetries gives chiral symmetry\aidx{chiral symmetry}
  \item Writing
\end{itemize}

\begin{equation}
\begin{aligned}
& P_{L}=\frac{1}{2}\left(1-\gamma_{5}\right), \qquad P_{R}=\frac{1}{2}\left(1+\gamma_{5}\right) \\
& \psi \rightarrow \exp \left(-\frac{i}{2} \alpha_{a} T_{a} P_{L, R}\right) \psi, \qquad \bar{\psi} \rightarrow \bar{\psi} \exp \left(\frac{i}{2} \alpha_{a} T_{a} P_{R, L}\right)
\end{aligned}
\end{equation}

gives \(G_{L} \times G_{R} \times U_{V}(1) \times U_{A}(1)\) global symmetry

\begin{itemize}
  \item Writing
\end{itemize}

\begin{equation}
\begin{aligned}
& \psi=\psi_{L}+\psi_{R}=P_{L} \psi+ P_{R} \psi \\
& \bar{\psi}=\psi^{\dagger} \gamma_{0}=\left(\psi^\dagger P_{L}+\psi^\dagger P_{R}\right) \gamma_{0}=\bar{\psi}\left(P_{R}+P_{L}\right)=\bar{\psi}_{L}+\bar{\psi}_{R}
\end{aligned}
\end{equation}

then

\begin{equation}
\begin{aligned}
\bar{\psi} \slashed{D} \psi & =\left(\bar{\psi}{P_{L}}+\bar{\psi} P_{R}\right) \slashed{D}\left(P_{L} \psi+P_{R} \psi\right) \\
& =\bar{\psi}_{L} \slashed{D} \psi_{L}+\bar{\psi}_{R} \slashed{D} \psi_{R}
\end{aligned}
\end{equation}

And other terms vanish, showing the two chiralities are independent\\

\section{Naive lattice fermions \& fermion doubling}

\begin{itemize}
  \item The continuum gauge invariant action for a single fermion can be discretised as
\end{itemize}

\begin{equation}
\begin{aligned}
S_{\text {naive }} & =a^{4} \sum_{n}\left[\frac{1}{2 a } \sum_{\mu} \bar{\psi}(n) \gamma_{\mu}\left\{U_{\mu}(n) \psi(n+\hat{\mu})-U_{\mu}^{\dagger}(n-\hat{\mu}) \psi(n-\hat{\mu})\right\}+M \bar{\psi}(n) \psi(n)\right] \\
& =a^{4} \sum_{n, m \in \Lambda} \bar{\psi}(n) D(n, m) \psi(n)
\end{aligned}
\end{equation}

where \(D(n, m)\) is a matrix in spacetime, color, Dirac indices

\begin{equation}
D(n,m)_{b \beta}^{a \alpha}=\frac{1}{2 a} \sum_{\mu}\left(\gamma_{\mu}\right)_{\alpha \beta}\left[U_{\mu}(n)_{a b} \delta_{n+\hat{\mu}, m}-U_{\mu}^{\dagger}(n)_{a b} \delta_{n-\hat{\mu}, m}\right]+M \delta_{\alpha \beta} \delta_{\alpha b} \delta_{nm}
\end{equation}

NB: used cental difference so \(D(n, m)\) is anti-Hermitian as \(\partial_{\mu} \sim i k_{\mu}\) is

\begin{itemize}
  \item This naive discretisation has a peculiar continuum limit\aidx{continuum limit} 
\end{itemize}

Consider the Fourier transform of the inverse of the free two point function (fermion propagator\aidx{fermion propagator})

\begin{equation}
\begin{aligned}
 \tilde{D}(k, l)&=\sum_{m, n} e^{-i(k \cdot n-l \cdot m)}\langle\Omega| \psi(n) \tilde{\psi}(m)|\Omega\rangle^{-1} \\
& =\sum_{m,n} e^{-i(k \cdot n \cdot - l \cdot m)} D(n,m | U_\mu = 1) \\
& = \sum_{m,n} e^{-i(k \cdot n - l \cdot m)}\left[ \frac{1}{2 a} \sum_\mu \gamma_\mu (\delta_{n + \hat{\mu}, m} - \delta_{n - \hat{\mu}, m}) + M \mathbb{I} \delta_{n,m}\right]\\
& = \sum_n e^{- i (k - l)\cdot n} \left( \frac{1}{2a}\sum_\mu \gamma_\mu ( e^{i l_\mu a}- e^{- i l_\mu a}) + M \mathbb{I}\right)\\
& = \delta^4(k-l)\tilde{D}(k), 
\end{aligned}
\end{equation}
where
\begin{equation}
\begin{aligned}
\tilde{D}(k) &= M + \frac{i}{a}\sum_\mu \gamma_\mu \sin k_\mu a\\
\tilde{D}^{-1}(k) &= \frac{M - \frac{i}{a}\sum_\mu \gamma_\mu \sin (k_\mu a)}{M^2 + \frac{1}{a^2} \sum_\mu \sin^2(k_\mu a)}
\end{aligned}
\end{equation}

Recall that we did this same calculation for the scalar field!

From this, the position space propagator $D(n,m)^{-1}$ can be determined by Fourier transformation

\begin{itemize}
  \item Setting \(M=0\), it is clear that \(\tilde{D}^{-1}(k)\) is singular of \(k_{\mu}=(0,0,0,0)\) as expected for a massless particle. However, it is also singular at $k_\mu^{(A)} \in \{ (0,0,0,\pi/a), \ldots,(\pi/a, \pi/a, \pi/a, \pi/a)\} \subset \mathcal{B}$, where $A = 0, 1, ..., 4, 12, ..., 123,..., 1234$ and $\mathcal{B}=\{p_\mu | - \pi/a < p_i \leq \pi/a \; \forall i\}$ is the Brillouin zone\aidx{Brillouin zone}  (BZ)
\end{itemize}

That is, there are \(16=2^{D}\) points about which there is an approximately linear dispersion relation - $15$ unwanted fermion doublers

Taking the naive $a \rightarrow 0$ limit near any $k^{(A)}$
\begin{equation}
    \tilde{D}^{-1}(k) = \frac{m - i \gamma_\mu p_\mu \cos(a k_\mu^{(A)})}{m^2 + p^2} + \mathcal{O}(a),\: k_\mu = k_\mu^{(A)} + p_\mu,\: p_\mu \ll \frac{1}{a}
\end{equation}

The \(\cos \left(a k_{\mu}^{(A)}\right)\) term is $\pm 1$ depending on A, and defining

\begin{equation}
\left\{S_{A}\right\}=\left\{1, S_{1}, \ldots S_{4}, S_{1} S_{2}, \ldots S_{1} S_{2} S_{3} \ldots S_{1} S_{2} S_{3} S_{4}\right\} \quad \text { for } S_{\rho}=i \gamma_{\rho} \gamma_{5}
\end{equation}

we can relate the different near-zero energy regions by a unitary transformation

\begin{equation}
\tilde{D}^{-1}(k)=S_{A}^{\dagger} \frac{m-i \gamma_{\mu} p_{\mu}}{m^{2}+p^{2}} S_{A}+\mathcal{O}\left(a^{2}\right) \quad k_{\mu}=k_{\mu}^{(A)}+p_{\mu} \quad p_{\mu}\ll \frac{1}{a}
\end{equation}

\begin{itemize}
  \item An interesting relation between doubling and chiral symmetry\aidx{chiral symmetry} is apparent if we consider the naive action for \(L\) and \(R\) handed Weyl fermions\aidx{fermions!Weyl} in the massless limit
\end{itemize}

\begin{equation}
\tilde{D}^{-1}(k)=\tilde{D}_{L}^{-1}(k)+\tilde{D}_{R}^{-1}(k)
\end{equation}

with

\begin{equation}
\begin{aligned}
 \tilde{D}_{L / R}^{-1}(k)&=S_{A}^{\dagger}\left[-\frac{i\gamma_{\mu} p_{\mu}}{p^{2}}\right] S_{A}\left(1 \pm \gamma_{5}\right) / 2 \\
& =S_{A}^{+}\left[\frac{-i \gamma_{\mu}p_\mu}{p^{2}}\left(1 \pm \varepsilon_{A} \gamma_{s}\right) / 2\right] S_{A} \\
& \text { where } \epsilon_{A}=\left\{\begin{array}{lll} 
+1 & A=12,13,34,\ldots,1234,0 & 8 \text { combos } \\
-1 & A=1,2,3,4,123, \ldots 234 & 8 \text { combos }
\end{array}\right.
\end{aligned}
\end{equation}

Meaning that the 16 doubler modes appear as $8$ LH and $8$ RH modes

\begin{itemize}
  \item More generally we can consider a fermion action with LH chirality
\end{itemize}

\begin{equation}
S_{F}=a^{4} \sum_{m, n} \bar{\psi}_{n} \gamma_{\mu} F_{\mu}(m, n)\left(1-\gamma_{5}\right) \psi_{m}
\end{equation}

for which
\begin{equation}
    \begin{aligned}
        &\text{(1) Translationally invariant: \(F_{\mu}(m, n)=F_{\mu}(m-n) \Rightarrow\) diagonal in $p$-space}\\
&\text{(2) Local (Fourier transform is smooth)}\\
&\text{(3) Hermitian action \(F_{\mu}^{*}=-F_{\mu}\) (Fourier transform is real)}
    \end{aligned}
    \label{eq:NNreq}
\end{equation}

Then by the \underline{Nielsen-Ninomiya theorem\aidx{Nielsen-Ninomiya theorem}} (Nielsen \& Ninomiya NPB 193(1981) 173, Karsten PLB 192 (1981) 315, Karsten \& Smit NP 13183 (1981) 103), the action corresponds to an equal number of left and right handed modes at low energy

Following Itzykson \& Drouffe, label the zeros of $\tilde F_\mu$ (the Fourier transform of $F_\mu$) as $\bar{k}^{(\alpha)}$, which we can take to be ioslated as otherwise they do not correspond to the correct continuum limits, 

\begin{equation}
    \tilde F_\mu(k = \bar{k}^{(\alpha)}+ p) = \underset{\in \mathbb{R}}{\underbrace{\mathcal{M}_{\mu \nu}^{(\alpha)}}} p_\nu \text{ for }p \ll 1/a
\end{equation}

Since \(\mathcal{M}_{\mu \nu}^{(\alpha)} \in \mathbb{R}\) we can write \(\mathcal{M}_{\mu \nu}^{(\alpha)}=O_{\mu \rho}^{(\alpha)} S_{\rho \nu}^{(\alpha)}\), where $O$ is an orthogonal matrix and $S$ is a positive symmetric matrix.

$O_{\mu \nu}^{(\alpha)} \in O(D)$ can be promoted to a $(D+1)$-dimensional rotation which acts on the first $D$ coordinates as a rotation and for the last coordinate it multiplies by $\operatorname{det} O_{\mu \nu}^{(\alpha)} = \pm 1$ (the index of $\tilde{F}_\mu$ at $\bar k^{(\alpha)}$).
%
This implies that $O_{\mu \nu}^{(\alpha)} \in SO(D+1)$, and the spinor representation of $SO(D+1)$ is given by $(\gamma_1, \ldots, \gamma_D, \gamma_5)^T$.

There exists a unitary $\Lambda^{(\alpha)}$ such that
\begin{equation}
    \Lambda^{(\alpha)} \gamma_\mu \Lambda^{(\alpha)^{-1}} = O_{\mu \nu}^{(\alpha)} \gamma_\nu
\end{equation}
and
\begin{equation}
    \Lambda^{(\alpha)} \gamma_5 \Lambda^{(\alpha)^{-1}} = \operatorname{det}(O_{\mu \nu}^{(\alpha)}) \gamma_5 = \operatorname{det}(O^{(\alpha)})\gamma_5
\end{equation}

Thus the action takes the form (in momentum space)
\begin{align}
    S_F &\sim \sum_\alpha \int \frac{\text{d}^4p}{(2 \pi)^4}\bar{\psi}(-k^{(\alpha)}-p) \gamma_\mu O_{\mu \nu}^{(\alpha)}S_{\nu \rho}^{(\alpha)}p_\rho^{(\alpha)}(1 - \gamma_5)\psi(k^{(\alpha)}+p) \nonumber \\
    &\sim \sum_\alpha \int_p \underset{\bar{\tilde{\psi}}(-k^{(\alpha)}-p)}{\underbrace{\bar{\psi}(-k^{(\alpha)}-p) \Lambda^{(\alpha)} }} \gamma_\mu S_{\nu \rho}^{(\alpha)} p_\rho^{(\alpha)} (1 - \operatorname{det}(O_{\mu \nu}^{(\alpha)}) \gamma_5)\underset{\tilde{\psi}(k^{(\alpha)}+p)}{\underbrace{\Lambda^{(\alpha)^{-1}}\psi(k^{(\alpha)}+p)}}
\end{align}

and since $S_{\nu \rho}^{(\alpha)}$ can be absorbed by a rescaling of $p_\rho^{(\alpha)}$, the net effect is the modification of the LH projector $(1 - \gamma_5) \rightarrow (1 - \operatorname{det}O_{\mu \nu}^{(\alpha)} \gamma_5)$ depending on the sign of $\operatorname{det}O_{\mu \nu}^{(\alpha)}$. Note that the above equation is somewhat schematic regarding the whole action - you might worry about there being double counting in the summation over $\alpha$, but the focus is on regions near the zeros. 

By the Poincare-Hopf index theorem\aidx{Poincare-Hopf index theorem}, $\sum_\alpha \operatorname{det}O^{(\alpha)}$ is the Euler character\aidx{characters (group)}istic of thecompact manifold on which $F_\mu(n)$ is defined. This manifold is $T^4$ ($\mathcal{B}$ is a $4-$torus) which has Euler characteristic $\chi(T^4) = 0$.

This implies that there are an equal number of $\alpha$ with $\operatorname{det}O^{(\alpha)} = +1$ and $-1$, and in particular it is not possible that there is a single $\alpha$ (point at which action vanishes).

Therefore, we cannot construct an action for a single chiral fermion satisfying the requirements in Eq.~\eqref{eq:NNreq}.

The above relied on $D+1$ being odd - chirality only makes sense in this case.

This is following David Tong's lectures
\begin{itemize}
    \item Consider a single Weyl fermion with general Hamiltonian in $d=3$ spatial dimensions
\end{itemize}

\begin{equation}
H=\vec{v}(\vec{k}) \cdot \vec{\sigma}+\varepsilon(\vec{k})
\end{equation}

There are massless modes when $v_i(\vec{k}) = 0, \forall i = 1,2,3$ -- this happens at $P$ isolated points $\vec{k}_\alpha, \alpha = 1,\ldots, P$. Whether each massless mode is L or R going depends on $\operatorname{sign}(\operatorname{det}\frac{\partial v_i}{\partial k_j}|_{k = k_\alpha})= \text{ chirality.}$

\begin{itemize}
  \item For \(\vec{k} \neq \vec{k}_{\alpha}\), the unit vector \(\hat{v}(\vec{k})=\frac{\vec{v}(\vec{k})}{|\vec{v}(\vec{k})|} \in C P_{\alpha}^{1} \sim S_{\alpha}^{2}\) (where $\alpha$ just labels the particular zero) which varies over angles defining $\vec{k}$, defining a map
\end{itemize}

\begin{equation}
\hat{v}(\vec{k}): S^{2} \rightarrow S_{\alpha}^{2}
\end{equation}

It is possible there is nontrivial winding, characterised by the second homotopy group

\begin{equation}
\pi_{2}\left(S_{\alpha}^{2}\right)=Z
\end{equation}

and while all \(n \in \mathbb{Z}\) can occur, case \(n \neq \pm 1\) can be separated by slight perturbation of $\hat{v}(\vec{k})$

\begin{itemize}
  \item The winding is equivalent to the chirality and is given by
\end{itemize}

\begin{equation}
v_{\alpha}=\frac{1}{8 \pi} \int_{S_{\alpha}^{2}} d^{2} S_{i} \varepsilon^{i j k} \varepsilon^{a b c} v^{a} \frac{\partial v^{b}}{\partial k^{j}} \frac{\partial v^{c}}{\partial k^{k}}
\end{equation}

for a vector field \(v\). TODO: NEED TO EXPLAIN THIS BETTER

\begin{itemize}
  \item Since the Brillouin zone is compact so gives no contribution to winding.
\end{itemize}
Writing ${\cal B}^\prime = {\cal B} \backslash\sum_\alpha S_\alpha^2$ i.e. the boundary of ${\cal B}^\prime$ is $\partial({\cal B}^\prime) = \sum_\alpha S_\alpha^2$,

\begin{align}
    \sum_\alpha v_\alpha &= \sum_\alpha \int_{S_\alpha^2}\text{d}^2S_i x_i \nonumber \\
    &= \int_{\partial({\cal B}^\prime)}\text{d}S_i x_i \nonumber \\
    &= \int_{{\cal B}^\prime}\text{d}^3 k \nabla_i \cdot x_i = \int \text{d}^3 k \frac{\partial}{\partial k_i}\underset{=0}{\underbrace{(\varepsilon^{ijk}\varepsilon^{abc}v^a \frac{\partial v^b}{\partial k_j}\frac{\partial v^c}{\partial k_k})}} \nonumber\\
    &= 0.
\end{align}
Thus the number of $+1$ windings (R chirality) and the number of $-1$ windings (L chirality) must be the same.
\section{ Wilson fermions }

\begin{itemize}
  \item To solve the doubling problem\aidx{fermion doubling}, Wilson fermions\aidx{fermions!Wilson} effectively introduce a momentum dependent mass that vanishes for \(k=(0,0,0,0)\) but \(\sim 1 / a\) for the other doubler modes
\end{itemize}

\begin{align}
    S_W &= a^4 \sum_{m,n}\bar{\psi}_m D_W(n,m)\psi_n \nonumber \\
    D_W(n,m) =& M \delta_{mn} + \frac{1}{2a}\sum_\mu \gamma_\mu (U_\mu(n) \delta_{n + \hat{\mu}, m} - U^\dagger_\mu(n - \hat{\mu}) \delta_{n - \hat{\mu}, m}))\nonumber \\
    &+ \frac{a r}{2} \underset{\nabla^2}{\underbrace{\frac{1}{2 a^2}\sum_\mu(U_\mu(n) \delta_{n + \hat{\mu}, m} - 2 \delta_{n,m} + U^\dagger_\mu(n - \hat{\mu})\delta_{n - \hat{\mu},  m})}},
\end{align}
where $0 \leq r \leq 1$ is a free parameter. $r$ outside this range gives a complex dispersion relation; for $r=1$, $S_W$ is reflection positive\aidx{reflection positivity}.

\begin{itemize}
    \item For $r=1$, the action simplifies using $P_\mu^{\pm} = \frac{1}{2}(1 \pm \gamma_\mu)$ (for $r \neq 1$, the $P^{\pm}_\mu$ are nor orthogonal)
\end{itemize}

\begin{equation}
D_{W}^{r=1}(n, m)=(M+4) \delta_{n, m}+\frac{1}{2 a} \sum_{\mu}\left(P_{\mu}^{+} U_{\mu}(n) \delta_{n+\hat{\mu}, m}+P_{\mu}^-U_{\mu}^{\dagger}(n-\hat{\mu}) \delta_{n-\hat{\mu}, m}\right)
\end{equation}

\begin{itemize}
  \item Rescaling the quark fields \(\psi_{n} \rightarrow \sqrt{M+4} \psi_{n}, \bar{\psi}_{n} \rightarrow \sqrt{M+4} \bar{\psi}_{n}\), the action can be written as
\end{itemize}

\begin{equation}
    \boxed{D_W^{r=1}(n,m) = \delta_{n,m} + \kappa \sum_\mu (P_\mu^+ U_\mu(n)\delta_{n + \hat{\mu}, m} + P_\mu^{-}U^\dagger_\mu(n - \hat{\mu}) \delta_{n - \hat{\mu}, m})}
\end{equation}

where the hopping parameter\aidx{kappa@$\kappa$ (hopping parameter)} \(\kappa=\frac{1}{2(M+4 r)}\) (for bare \(M=0, \kappa=1 / 8 r\) )

\begin{itemize}
  \item The inverse free propagator\aidx{scalar field!propagator} in momentum space
\end{itemize}

\begin{equation}
\widetilde{D}_W(k)=M+\frac{r}{a} \sum_{\mu}\left[1-\cos \left(k_{\mu} a\right)\right]+\frac{i}{a} \sum_{\mu} \gamma_{\mu} \sin \left(k_{\mu} a\right)
