\section{Chiral symmetry \& lattice fermions}

\begin{itemize}
  \item Approximate chiral symmetry is crucial to phenomenology of QCD. For $N_f = 3$ ``light'' flavours of quarks $(m_f < \Lambda_\text{QCD} \sim \text{typical QCD scale} \sim m_\text{proton})$ there is an approximate
\end{itemize}

\begin{equation}
S U_{L}\left(N_{f}\right) \times \operatorname{SU}_{R}\left(N_{f}\right) \times U_{V}(1) \times U_{A}(1)
\end{equation}

symmetry of the lagrangian. Quark masses break the symmetry explicitly but even for \(m_{q}=0\), the strong dynamics of QCD spontaneously breaks the symmetry

\begin{equation}
S U_{L}\left(N_{f}\right) \times S U_{R}\left(N_{f}\right) \times U_{V}(1) \times U_{A}(1) \rightarrow S U_{V}\left(N_{f}\right) \times U_{V}(1)
\end{equation}

\begin{itemize}
  \item Spontaneous symmetry breaking is characterised by a nonzero expectation value for the chiral condensate
\end{itemize}

\begin{equation}
\sum_{q}\langle\bar{q} q\rangle \neq 0
\end{equation}

\begin{itemize}
    \item SSB leads to the appearance of Goldstone bosons which would be massless if $m_q \neq 0$ but remain very light even for $m_q^\text{phys}$. In nature, the pseudoscalar meson octet $(\pi^0, \pi^+, \pi^-, K^-, K^0, \bar{K}^0, K^+, \eta)$ are very light compared to typical hadrons such as the proton
    \item Although the QCD action has an axial $U_A(1)$ symmetry classically, this is broken by quantum effects (anomalous) since the path integral measure is not invariant (the Adler-Bell-Jackiw anomaly)
\end{itemize}

\begin{equation}
\partial_{\mu} J_{A}^{\mu}=-2 i m \bar{\psi}{\gamma_{5}} \psi+\frac{g^{2}}{16 \pi^{2} }\varepsilon_{\mu \nu \rho \sigma} F_{\mu \nu} F_{\rho \sigma},
\end{equation}
where the last term is anomalous and can be derived using Fujikawa's method.

\begin{itemize}
  \item It is important that lattice discretisations are able to reproduce this physics
  \item The NN theorem says that it is not possible to construct a fermion discretisation that is both chiral (and local) and doubler free
  \begin{itemize}
      \item Wilson removed doublers but explicitly broke chiral symmetry
      \item Staggered fermions reduced doublers ($2^{D/2}$ rather than $2^D$) and maintained some chiral symmetry
  \end{itemize}
  \item Another way around NN no-go theorem is to redefine chiral symmetry
  \item Based on RG blocking, Ginsparg and Wilson noted that chiral symmetry would be recoverd in the continuum if at finite lattice spacing
\end{itemize}

\begin{equation}
\boxed{\left\{D, \gamma_{5}\right\}=a D \gamma_{5} D},
\end{equation}
where $D$ is a lattice Dirac operator. This is known as the Ginsparg-Wilson relation.

\begin{itemize}
  \item Requiring the GW relation rather than \(\left\{D, \gamma_{5}\right\}=0\) allows a chiral symmetry to be defined at \(a \neq 0\). The appropriate symmetry was found by L\"uscher: for a Dirac spinor
\end{itemize}

\begin{equation}
\boxed{\psi \rightarrow \psi^{\prime}=\exp \left(i \alpha \gamma_{5}\left(1-\frac{a}{2} D\right)\right) \psi \quad \bar{\psi} \rightarrow \bar{\psi}^{\prime}=\bar{\psi} \exp \left(i \alpha\left(1-\frac{a}{2} D\right) \gamma_{5}\right)}
\end{equation}

\begin{itemize}
  \item The action is invariant
\end{itemize}
\begin{equation}
\begin{aligned}
S=\int \bar{\psi} \slashed{D} \psi \rightarrow \int \bar{\psi}^{\prime} D \psi^{\prime} & =\int \bar{\psi} e^{i \alpha\left(1-\frac{a}{2} D\right) \gamma_{5}} D e^{i \alpha \gamma_{5}\left(1-\frac{a}{2} D\right)} \psi \\
& =\int \bar{\psi} e^{i \alpha\left(1-\frac{a}{2}D\right) \gamma_{5}} e^{-i \alpha\left(1-\frac{a}{2} D\right) \gamma_{5}} D \psi  \\
& =\int \bar{\psi} D \psi
\end{aligned}
\end{equation}
where we have used from GW that
\begin{equation}
    D \gamma_5 \left(1- \frac{a}{2}D\right) + \left(1 - \frac{a}{2}D\right)\gamma_5 D = 0
\end{equation}
\begin{itemize}
  \item Defining modified chiral projectors as
\end{itemize}

\begin{equation}
\hat{P}_{L}=\frac{1}{2}\left(1-\hat{\gamma}_{5}\right), \hat{P}_{R}=\frac{1}{2}\left(1+\hat{\gamma}_{5}\right), \hat{\gamma}_{5}=\gamma_{5}(1-a D), \hat{\gamma}_{5}^{2}=1,
\end{equation}
we note that we have
\begin{equation}
    \hat{P}_L^2 = \hat{P}_L, \hat{P}^2_R = \hat{P}_R, \hat{P}_L + \hat{P}_R = 1, \hat{P}_R \hat{P}_L = \hat{P}_L \hat{P}_R,
\end{equation}
which one can find again using GW. This implies then that
\begin{equation}
    D \hat{P}_R = P_L D, \quad D \hat{P}_L = P_R D \quad 
\end{equation}
with the usual $P_{R/L} = (1\pm\gamma_5)/2$.
It is natural to define

\begin{equation}
\psi_{R}=\hat{P}_{R} \psi, \psi_{L}=\hat{P}_{L} \psi \text { but } \bar{\psi}_{L}=\bar{\psi} P_{R}, \bar{\psi}_{R}=\bar{\psi} P_{L}
\end{equation}

So

\begin{equation}
\begin{aligned}
\bar{\psi} D \psi=\bar{\psi}\left(P_{L}+P_{R}\right) D\left(\hat{P}_{L}+\hat{P}_{R}\right) \psi & =\bar{\psi} P_{L} D \hat{P}_{R} \psi+\bar{\psi} P_{R} D \hat{P}_{L} \psi \\
& =\bar{\psi}_{R} D \psi_R +\bar{\psi}_{L} D \psi_{L}
\end{aligned}
\end{equation}

\begin{itemize}
  \item This trivially generalises to \(N_{f}\) flavors of fermions whereby the vector transformations are as before, but axial transformations are
\end{itemize}

\begin{equation}
\begin{aligned}
& \psi \rightarrow \exp \left(i \alpha_{a} T^{a} \gamma_{5}(1-a D / 2)\right) \psi \\
& \bar{\psi} \rightarrow \bar{\psi} \exp \left(i \alpha_{a} T^{a}(1-a D / 2) \gamma_{5}\right)
\end{aligned}
\end{equation}

and the full \(S U_{L}\left(N_{f}\right) \times S U_{R}\left(N_{f}\right) \times U_{1}(V)\) symmetry is maintained (the \(U_{A}(1)\) symmetry is anomalous as in the continuum)

\begin{itemize}
  \item Having defined \(\psi_{L, R}\) and \(\bar{\psi}_{L, R}\) above, the natural mass term (mimicking the continuum) is
\end{itemize}

\begin{equation}
m\left(\bar{\psi}_{L} \psi_{R}+\bar{\psi}_{R} \psi_{L}\right)=m \bar{\psi}\left(P_{R} \hat{P}_{R}+P_{L} \hat{P}_{L}\right) \psi=m \bar{\psi}\left(1-\frac{a}{2} D\right) \psi
\end{equation}

\begin{itemize}
  \item The GW relation can be generalised to
\end{itemize}

\begin{equation}
\left\{\gamma_{5}, D\right\}=a D {\gamma_5} R D
\end{equation}

for a local operator \(R\) (strictly ultra-local = exponentially localised)\\
Immediately this implies

\begin{equation}
D+D^{\dagger}=a D^{\dagger} R D \quad \text { where } D^{\dagger}=\gamma_{5} D \gamma_5
\end{equation}

\begin{itemize}
  \item A Dirac operator satisfying GW relation is \(\gamma_{5}\)-Hermitian: \(\gamma_{5} D \gamma_{5}=D^{\dagger}\) The operator \(H=\gamma_5 D\) is Hermitian
  \item Note that the chiral transformations above are slightly non-local, involving at least nearest neighbour. From the GW relation [act on \(L, R\) with \(D^{-1}\) ]
\end{itemize}

\begin{equation}
\left\{\gamma_5, D^{-1}(x, y)\right\}=a \gamma_{5} \delta(x-y)
\end{equation}

which means that the fermion propagator only violates chirality locally

\begin{itemize}
  \item Spectrum of $D$
  \item The massless continuum Dirac operator is anti-Hermitian \(\slashed{D}^{\dagger}=-\slashed{D}\) and so has purely imaginary eigenvalues and orthonormal eigenvectors (\(L=R\) eigenvectors)
\end{itemize}

\begin{equation}
\slashed{D}\left|v_{\lambda}\right\rangle=\lambda\left|v_{\lambda}\right\rangle, \lambda=-\lambda^{*}
\end{equation}

Also, since \(\slashed{D} \gamma_5 = - \gamma_5 \slashed{D}, \quad \gamma_5 |v_\lambda\rangle\) is an eigenvector with eigenvalue $-\lambda$
\begin{equation}
    \slashed{D}\gamma_5 |v_\lambda \rangle = - \gamma_5 \slashed{D} |v_\lambda\rangle = - \lambda \gamma_5 |v_\lambda\rangle
\end{equation}

and non-zero eigenvalues are paired\\
For the zero modes $\slashed{D}|v_0^{(i)}\rangle = 0$, $\slashed{D}\gamma_5 | v_0^{(i)}\rangle = 0$ so $\gamma_5|v_0^{(i)}\rangle$ is also a zero mode and in this subspace we can choose zero modes such that they are eigenstates of \(\gamma_5\) too

\begin{equation}
\gamma_{5}\left|v_{0}^{(i)}\right\rangle= \pm\left|v_{0}^{(i)}\right\rangle
\end{equation}

(eigenvalues \(= \pm 1\), since \(\gamma_{5}^{2}=1\) and \(\gamma^\dagger_5=\gamma_5\) )

\begin{itemize}
  \item For a lattice Dirac operator obeying the GW relation, \(\gamma_{5}\)-hermiticity constrains the spectrum to have real or complex conjugate eigenvalues. Eigenvalues of \(D\) are determined by
\end{itemize}
\begin{equation}
    P(\lambda) = \operatorname{det}(D-\lambda I) = \operatorname{det}(\gamma_5^2D - \lambda I) = \operatorname{det}(D^\dagger - \lambda I) = \operatorname{det}(D - \lambda^* I)^* = P^*(\lambda^*),
\end{equation}
and since the condition for an eigenvalue is $P = 0$, this implies that if $\lambda$ is an eigenvalue, so is $\lambda^*$.

\begin{itemize}
  \item The GW relation guarantees that \(D\) is normal: \(\left[D, D^{\dagger}\right]=0\)
\end{itemize}

\begin{equation}
\begin{array}{rcl}
\gamma_{5}\left(D\gamma_5+\gamma_{5} D\right)=\gamma_{5}\left(a D{\gamma_{5}} D\right) & \qquad : \qquad & \left(D{\gamma_{5}}+\gamma_{5} D\right) \gamma_{5}=\left(a D{\gamma_{5}} D\right){\gamma_{5}} \\
D^{\dagger}+D=a D^{\dagger} D &\qquad  : \qquad & D+D^{\dagger}=a D D^{\dagger} \\
& \Rightarrow D^{\dagger} D=D D^{\dagger} &
\end{array}
\end{equation}

Normality is sufficient for the spectral theorem to guarantee that the eigenvectors form an orthonormal basis \(\left\{e_{\lambda}\right\}\)

\begin{itemize}
  \item From GW,
\end{itemize}
\begin{equation}
    0 = (D^\dagger + D - a D^\dagger D) e_\lambda = (\lambda^* + \lambda - a \lambda^* \lambda)e_\lambda,
\end{equation}
and writing $\lambda = x+iy$ this yields
\begin{equation}
    \left(x-\frac{1}{a}\right)^2 + y^2 = \frac{1}{a^2}.
\end{equation}
This is the equation of a circle on which the doubler modes must all live, shown in Fig.~\ref{fig:doubler_circle}. The only real solutions are $\lambda = 0, \frac{2}{a}$ and doubler modes all live at $2/a$.

    \includegraphics[width=0.5\linewidth]{fig-notes-076.png}
    \caption{The possible eigenvalues of $D$ assuming GW holds.}
    \label{fig:doubler_circle}
\end{figure}

\begin{itemize}
  \item Chirality of modes: the zero modes \(\left\{e_{0}^{(i)} \mid D e_{0}^{(i)}=0\right\}\) can be chosen to be eigenvectors of \(\gamma_5\) as in the continuum
\end{itemize}

\begin{equation}
\gamma_{5} e_{0}^{(i)}= \pm e_{0}^{(i)}
\end{equation}

For other modes, note that

\begin{equation}
\lambda\left(e_{\lambda}, \gamma_{5} e_{\lambda}\right)=\left(e_{\lambda}, \gamma_{5} D e_{\lambda}\right)=\left(e_{\lambda}, D^{\dagger} \gamma_{5} e_{\lambda}\right)=\left(D e_{\lambda}, \gamma_{5} e_{\lambda}\right)=\lambda^{*}\left(e_{\lambda}, \gamma_{5} e_{\lambda}\right)
\end{equation}

so if \(\lambda \neq \lambda^{*}\) then \(\left(e_{\lambda}, \gamma_{5} e_{\lambda}\right)=0\)\\
Consequently, defining
\begin{align}
    Q &\equiv \frac{a}{2}\operatorname{tr}[\gamma_5 D] = -\frac{1}{2}\operatorname{tr}[\gamma_5(2-aD)] \nonumber\\
    &= -\frac{1}{2}\sum_\lambda (e_\lambda, \gamma_5(2-aD)e_\lambda) \nonumber\\
    &=-\frac{1}{2}\sum_\lambda(2-a\lambda)(e_\lambda, \gamma_5 e_\lambda) \nonumber\\
    &=-\frac{1}{2}\sum_i 2(e_0^{(i)}, \gamma_5e_0^{(i)}) \nonumber\\
    &= n_- - n_+
\end{align}
where the trace is over the Dirac and spacetime indices, we were able to insert the $2$ in the first line since the trace of $\gamma_5$ is $0$, the third line simplifies dramatically because the inner product vanishes unless $\lambda$ is real (and therefore either $0$ or $\frac{2}{a}$) and the first term is chosen to vanish for $\lambda = \frac{2}{a}$, so only zero modes contribute (and the sum on the fourth line is over zero modes indexed by $i$). The fifth line then denotes the difference in the number of modes with $\gamma_5 e_0^{(i)} = \mp 1$.

This is remarkable since \(\gamma_{5} D\) is a complicated function of gauge fields in the interacting theory and depends on exactly the form of discretisation of \(D\) but the RHS is an integer!

\begin{itemize}
  \item An alternate derivation of this follows from the GW relation which equates
\end{itemize}

\begin{equation}
a(z-D) \gamma_{5}(z-D)=z(2-a z) \gamma_{5}-(1-a z)\left[(z-D) \gamma_{5}+\gamma_{5}(z-D)\right]
\end{equation}

Multiplying from the left with \((z-D)^{-1}\) and taking the trace over Dirac and spacetime indices (using the cyclic property of the trace) and dividing by \(z(2-a z)\) gives
\begin{equation}
    -\frac{a\operatorname{tr}[\gamma_5 D]}{(2-az)z} = \operatorname{tr}[\gamma_5(z-D)^{-1}]
\end{equation}
Now performing a contour integral around $z=0$ such that no spectral value of $D$ other than $0$ is included (using the fact that the LHS takes a pole at $z=0$),
\begin{equation}
    \Rightarrow -\frac{a}{2}\operatorname{tr}[\gamma_5 D] = \operatorname{tr}[\gamma_5 P_0]\text{ with }P_0 = \frac{1}{2 \pi i}\oint \frac{\text{d}z}{z - D}
\end{equation}

Here $P_0$ projects out the zero modes of $D$ and so $\operatorname{tr}[\gamma_5 P_0] = n_- - n_+$

\begin{itemize}
  \item The fermion integration measure is not invariant under the lattice chiral symmetry
\end{itemize}

\begin{equation}
\begin{aligned}
& \mathcal{D} \bar{\psi} \mathcal{D} \psi \rightarrow \mathcal{D} \bar{\psi}^{\prime} \mathcal{D} \psi^{\prime}=\mathcal{D} \bar{\psi}\left(1+i \alpha_{a} T_{a}\left(1-\frac{a}{2} D \gamma_{5}\right)\left(1+i \alpha_{a} T_{a} \gamma_{5}\left(1-\frac{a}{2} D\right)\right) \mathcal{D} \psi\right. \\
& =\mathcal{D} \bar{\psi} \mathcal{D} \psi\left(1+i \alpha_{a} \operatorname{Tr}\left[T_{a} \gamma_{5}(2-a D)\right]\right)+O\left(\alpha^{2}\right), \qquad a=0,1, \ldots, N_{f}^{2}-1 \\
& =\mathcal{D} \bar{\psi} \mathcal{D} \psi\left(1 - i a \alpha_{0} \operatorname{Tr}\left[\gamma_{5} D\right]\right), \quad\text{ since }\ \operatorname{tr}\left[T^{a}\right]=0, \quad\operatorname{tr}\left[\gamma_{5}\right]=0 \\
& =\mathcal{D} \bar{\psi} \mathcal{D} \psi\left(1-2 i \alpha_{0} Q\right)
\end{aligned}
\end{equation}

Thus for a quantity

\begin{equation}
\langle\theta\rangle=\frac{1}{2} \int \mathcal{D} \bar{\psi}\mathcal{D}\psi \theta[\psi, \bar{\psi}] e^{S},
\end{equation}
where the action $S$ obeys the GW relation, the variation under a flavour singlet axial transformation is

\begin{equation}
\langle\delta \theta\rangle=-a\operatorname{tr}\left[\gamma_{5} D\right]\langle\theta\rangle
\end{equation}

\begin{itemize}
    \item Just as in Fujikawa's method in the continuum, the ABJ anomaly arises from the measure rather than a change in the action. For Wilson fermions, it can be shown that the ABJ anomaly is reproduced via the variation of the action under chiral transformation (for which the measure is invariant). Note: Unlike in Fujikawa's method in the continuum, the set of eigenvalues of $D$ is finite in the lattice-regulated theory, so $\operatorname{Tr}[\gamma_5] = \sum_\text{modes}(\operatorname{tr}[\gamma_5] = 0)$. Note also that any symmetry that is exact on the lattice will be exact in the continuum limit, while any symmetry that is anomalous in the continuum must be broken on the lattice
\end{itemize}

\section{The overlap operator}

\begin{itemize}
  \item Neuberger (following work of Narayanan\& Neuberger) introduced the overlap Dirac operator that explicitly satisfies the GW relation
\end{itemize}

\begin{equation}
D_{O}=\frac{1}{a}\left(1+\gamma_{5} \operatorname{sign}(H)\right), \quad H=\gamma_{5} A
\end{equation}

where \(A\) is a \(\gamma_{5}\)-Hermitian kernel often based on the Wilson Dirac operator: \(A=a D_{W}-(1+s)\) with $s \in (-1, 1)$. Note that realizing this construction would work was very nontrivial.\\
This can also be written as
\begin{equation}
    D_O = \frac{1}{a}\left(1 + \gamma_5 H / \sqrt{H^2}\right),
\end{equation}
which is well-defined since $H^2 = \gamma_5 A \gamma_5 A = A^\dagger A$.

\begin{itemize}
  \item This satisfies GWR
\end{itemize}

\begin{align}
    a D_O \gamma_5 D_O \gamma_5 &= a D_O D_O^\dagger \nonumber \\
    &=\frac{1}{a}(1 + \gamma_5 \operatorname{sign}(H))(1 + \operatorname{sign}(H)\gamma_5) \nonumber \\
    &=\frac{1}{a}(1 + \gamma_5 \operatorname{sign}(H) + \operatorname{sign}(H) \gamma_5 + \gamma_5^2(\operatorname{sign}(H))^2) \nonumber \\
    &=\frac{1}{a}(1 + \gamma_5 \operatorname{sign}(H))+\frac{1}{a}(1 + \operatorname{sign}(H)\gamma_5) \nonumber \\
    &=D_O\gamma_5^2 + \gamma_5D_O \gamma_5 = \{\gamma_5,D_O\}\gamma_5\text{, as desired.}
\end{align}

\begin{itemize}
    \item The overlap operator looks non-local since it involves $(H^2)^{-1/2}$, but it is in fact ``local'' in the sense that for sufficiently smooth fields
\end{itemize}
\begin{equation}
    |D_O(n,m)| \leq C \exp{(-\gamma|n-m|)},
\end{equation}
where $\gamma$ is a fixed distance in lattice units, and in particular $C,\gamma > 0$ are independent of the gauge field. However, $D_O(n,m) \neq 0$ for any $|n-m|$. See Hernandez et al NPB 552(1999) 363.

\begin{itemize}
    \item If $D_W$ is used, then the overlap operator in momentum space is
\end{itemize}
\begin{align}
    a \tilde{D}_O(k) &= 1-[1-2\sum_\mu \sin^2(a k_\mu/2) - i \sum_\mu \gamma_\mu \sin(ak_\mu)][1+8\sum_{\mu<\nu}\sin^2(a k_\mu/2)\sin^2(a k_\nu)/2]^{-1/2}\nonumber \\
    &= 1 - [1 - i a \sum_\mu \gamma_\mu \overset{\circ}k_\mu - \frac{1}{2}a^2 \sum_\mu \hat{k}_\mu^2][1 + \frac{a^4}{2}\sum_{\mu < \nu}\hat{k}_\mu^2 \hat{k}_\nu^2]^{-1/2},
\end{align}
where $\overset{\circ}k_\mu \equiv \frac{1}{a}\sin(a k_\mu)$ and $\hat{k}_\mu = \frac{2}{a}\sin(\frac{a}{2}k_\mu)$. This is an analytic function of $k$ inside $\mathcal{B}$ and is periodic; it is invertible everywhere in $\mathcal{B}$ except at $k_\mu = (0,0,0,0)$, and for $k$ near $(0,0,0,0)$, $\tilde{D}_O(k) \sim i \gamma_\mu k_\mu + \mathcal{O}(a^2)$
\begin{itemize}
  \item The overlap operator maps the spectrum of its kernel operator onto the GW circle discussed above (see Fig.~\ref{fig:overlap_gw_circle})
  \item Numerically implementing \(D_{O}\) is non-trivial and expensive as the sign function must be approximated carefully

\end{figure}
\end{itemize}

\section{Domain Wall Fermions}

  \begin{itemize}
  \item Somewhat before the development of overlap fermions, Kaplan introduced domain-wall fermions that approximately satisfy the GW relation (they satisfy it exactly in a limit)
  \item The fundamental idea behind DWF is best explored first in the continuum. Consider a free fermion in $5$D Euclidean spacetime with coordinates $(x_\mu, s)$. This satisfies the Dirac equation
\end{itemize}
\begin{equation}
    D_5 \psi(x, s) = [\gamma_\mu \partial_\mu + \gamma_5 \frac{\partial}{\partial s} + M(s)] \psi(x,s) = 0
\end{equation}
where we have allowed the mass to be more general and depend on $s$ (note: this breaks Poincare symmetry in $5D$, but not in $4$D; also parity)

To be specific, take
\begin{equation}
    M(s) = \begin{cases}
        +m, \quad s > 0 \\
        -m, \quad s<0
    \end{cases} \quad\text{ with }m>0.
\end{equation}
A smoothed version is also possible, but we need the kink $M(\pm \infty) = \pm m$. Then, writing solutions as
\begin{equation}
\psi(x,s) = \sum_n(b_n(s)P_R + f_n(s)P_L) \psi_n(x)
\end{equation}
and setting
\begin{equation}
    [\gamma_\mu \partial_\mu + \mu_n]\psi_n = 0,
\end{equation}

This implies
\begin{align}
    D_5 \psi &= \sum_n\{ -\mu_n(b_n P_L + f_n P_R) + (b'_nP_R - f'_nP_L) + M(s)(b_n P_R + f_n P_L)\} \psi_n = 0 \nonumber \\
    &= \sum_n \{ P_L(-\mu_n b_n - f'_n + M f_n) + P_R(-\mu_n f_n + b'_n + Mb_n)\} \psi_n = 0 \nonumber \\
    &\Rightarrow \left(\frac{\text{d}}{\text{d}s} + M(s)\right)b_n = \mu_n f_n, \quad \left(-\frac{\text{d}}{\text{d}s} + M(s)\right)f_n = \mu_n b_n
\end{align}

\begin{figure}
\includegraphics[max width=0.5\textwidth]{fig-notes-019.jpg}
\caption{The function $M(s)$ in red and the massless mode in blue.}
\end{figure}

Since $m$ is the only scale, we expect $\mu_n \gtrsim m$ (and there is an infinite tower of such solutions), but we also see that there is a solution to the first equation for $\mu_n = 0$ with (note that the lower integration limit $s' = 0$ is arbitrary)
\begin{equation}
b_{0}(s)=N \exp \left[-\int_{0}^{s} M\left(s^{\prime}\right) d s^{\prime}\right]=N \exp [-m|s|]
\end{equation}
but no corresponding solution to the second equation since $\exp[+m|s|]$ is not normalisable. This single RH massless mode is localised around the defect at $s=0$ and since
\begin{equation}
    \gamma_\mu \partial_\mu \psi_0(x) = 0 \Rightarrow \psi_0(x) = e^{ipx}
\end{equation}
with $p^2 = 0$.

\begin{itemize}
    \item This construction required an infinite $5$th dimension but we can also consider the situation for a finite slab $-s_0 < s < s_0$ with periodic BCs $\psi(x, s+2s_0) = \psi(x,s)$. For
\end{itemize}
\begin{equation}
    M(s) = m\frac{s}{|s|},
\end{equation}
there is a kink at $s=0$ and also at $s=s_0$ with the opposite discontinuity; in this case there are both RH and LH solutions
\begin{figure}
\includegraphics[max width=0.5\textwidth]{fig-notes-020.jpg}
\caption{$M(s)$ and the zero modes $b_0$ and $f_0$ in the case where the fifth dimension has finite extent and periodic boundary conditions.}
\end{figure}\\
\begin{equation}
    b_0(s) = N \exp{[-\int_{-s_0}^s M(s')\text{d}s']}\quad \& \quad f_0(s) = N' \exp{[\int_{s_0}^s M(s')\text{d}s']}
\end{equation}
which are peaked around $s=0$ and $s=s_0$, respectively, and since $\int_{-s_0}^{s_0}M(s) \text{d}s = 0$ correspond to massless states.
\begin{itemize}
    \item Away from the free field limit the two modes $\psi_L = f_0(s) \psi_0(x)$ and $\psi_R = b_0(s)\psi_0(x)$ couple, leading to a mass
\end{itemize}
\begin{equation}
    \delta m \sim \int_{-s_0}^{s_0}f_0(s)b_0(s)\text{d}s \sim \exp{(-m s_0)}
\end{equation}
\begin{itemize}
    \item Looking at Green functions of the $5$D Dirac equation for a RH mode
\end{itemize}

\begin{equation}
D_{5} G\left(x, s_{1} ; y, s_{2}\right)=\left(\gamma_\mu D_{\mu}+\gamma_{5} \partial_{s}+M\right) G\left(x, s_{1} ; y, s_{2}\right)=\delta^{4}(x-y) \delta\left(s_{1}-s_{2}\right)
\end{equation}

with \(P_LG\left(x, s_{1}=0 ; y_{1}, s_{2}\right)=0\)\\
This is solved for \(s_{1}=s_{2}=0\) by [derived by L\"uscher, see hep-ph/0102028]
\begin{equation}
    G(x, s_1 = 0; y,s_2 = 0) = 2MP_LD^{-1}(x,y)P_R
\end{equation}
where
\begin{equation}
    D = M[1 - \gamma_5 \text{sign}(Q)], \qquad Q = \gamma_5(M - \gamma_\mu \partial_\mu)
\end{equation}
which is the $4$D overlap operator!
\begin{itemize}
    \item Now consider a $5$D lattice $\Lambda_{5D} = \Lambda \times Z_{N_5}$ with $N_5$ sites in the $5$th dimension (See Blum et al hep-lat/0007038). The DWF action is (this formulation follows Shamir and Furman\& Shamir)
\end{itemize}
\begin{equation}
    S_\text{DWF} = \sum_{n,m\in\Lambda}\sum_{s_1, s_2=0}^{N_5 - 1}\bar{\Psi}(n,s_1) D_\text{DWF}(n,s_1;m,s_2)\Psi(m,s_2)
\end{equation}
with
\begin{equation}
    D_\text{DWF}(n,s_1;m,s_2) = \delta_{s_1, s_2}D_W(n,m;M_5) + \delta_{n,m} D_5^{(\text{DW})}(s_1, s_2)
\end{equation}

where $D_W$ is the $4$D Wilson operator with mass $M_5$ ($[0 \leq M_5 \leq 2]$ for a single LH, RH pair - other values produce different numbers of light modes) constructed from gauge links living in $4$D (replicated for each $s$) and
\begin{equation}
    D_5^{(\text{DW})} = \delta_{s_1, s_2} - (1-\delta_{s_1, N_5 - 1})\delta_{s_1 + 1,s_2}P_L - (1 - \delta_{s_1, 0})\delta_{s_1-1, s_2}P_R + m[\delta_{s_1, N_5 - 1}\delta_{0, s_2}P_L + \delta_{s_1, 0}\delta_{s_2, N_5 - 1}P_R]
\end{equation}
Here, $m$ will turn out to be the light fermion mass and the $1-\delta$ terms block hopping to the boundary slices if $m=0$. $D_5^{(DW)}$ is independent of the gauge field.
\begin{itemize}
    \item For $m=0$, the spectrum contains one RH Weyl fermion near the $s=0$ boundary and a LH Weyl fermion near the $s = N_5 - 1$ boundary. The spectrum also contains a tower of massive states that extend across the $5$th dimension with masses $\mathcal{O}(M)$
    \item The $4$D fields we are interested in can be defined as
\end{itemize}
\begin{align}
    \psi(n) &= P_L \Psi(n,0) + P_R \Psi(n, N_5 - 1) \nonumber \\
    \bar{\psi}(n) &= \bar{\Psi}(n,N_5 - 1) P_L + \bar{\Psi}(n,0)P_R
\end{align}
so for example the scalar bilinear is built from modes coupling between boundaries
\begin{equation}
    \bar{\psi}(n)\psi(n) = \bar{\Psi}(n,N_5 - 1) P_L \Psi(n,0) + \bar{\Psi}(n,0)P_R \Psi(n, N_5-1)
\end{equation}
which is exactly the structure of the \(m\) term in \(D_{5}^{(\text{DW})}\)

\begin{itemize}
  \item Since the massive modes extending across the $5$th dimension are unwanted, Furman \& Shamir introduced bosonic Pauli-Villars fields to explicitly cancel them:
\end{itemize}
\begin{equation}
    Z_{DWF} = \int \mathcal{D}\bar{\Psi}\mathcal{D}\Psi\mathcal{D}U\mathcal{D}\bar{\Phi}\mathcal{D}\Phi \exp[-S_{DWF}[\bar{\Psi},\Psi,U]-S_{PF}[\bar{\Phi}, \Phi, U] - S_G[U]]
\end{equation}
where $\Phi, \bar{\Phi}$ are unphysical $5$D bosonic fields with spin, colour, flavour, and spacetime structure exactly as for $\Psi$, $\bar{\Psi}$. $S_G[U]$ is the gauge action (any version is ok) and
\begin{equation}
    S_{PF}[\bar \Phi, {\Phi}, U] = \sum_{n,m}\sum_{s_1, s_2}\bar{\Phi}(n,s_1)D_{PF}(n,s_1;m,s_2)\Phi(m,s_2)
\end{equation}
with
\begin{equation}
    D_{PF}(n,s_1;m,s_2) = \delta_{s_1, s_2}D_W(n,m;M_5) + \delta_{n,m}D_5^{(DW)}(s_1,s_2)|_{m = 1}
\end{equation}
which is the same as the fermions except $m=1$ in the $5$D part

\begin{itemize}
    \item Since the $\Phi, \bar{\Phi}$ are bosonic $\int \mathcal{D}\bar{\Phi} \mathcal{D}\Phi e^{-S_{PF}} = \operatorname{det}(D_{PF}[U])^{-1}$ which acts to cancel the equivalent large mass modes in $\int \mathcal{D}\bar{\Psi}\mathcal{D}\Psi e^{-S_{DWF}} = \operatorname{det}(D_{DWF}[U]$
\end{itemize}

\begin{itemize}
    \item More formally, Neuberger showed (Neuberger PRD 57 (1998)5417) that
\end{itemize}

\begin{equation}
\label{eq:det_rat}
\frac{\operatorname{det}\left(D_{\text {DWF }}[U]\right)}{\operatorname{det}\left(D_{P F}[U]\right)}=\operatorname{det}\left(\frac{1+\gamma_{5} \tanh \left(\frac{N_{5}}{2} H\right)}{2}\right)
\end{equation}
where $H$ is related to $\gamma_5 D_W$ (in a somewhat complicated way, but it would be exactly this if the $5$th dimension was continuous.
In the limit $N_5 \rightarrow \infty$, $\tanh \rightarrow \operatorname{sign}$, so the right hand side of Eq.~\eqref{eq:det_rat} goes to $\operatorname{det}(D_{O})$ which is the contribution of a single massless fermion.

\begin{itemize}
  \item To compute correlation functions in \(4 D\) which are those we care about, the relation of the $4$D fields to the $5$D fields must be used
\end{itemize}
\begin{equation}
    \langle \mathcal{O}(\psi, \bar{\psi}) \rangle = \frac{1}{Z_{DWF}} \frac{\delta}{\delta \eta} \ldots \frac{\delta}{\delta \bar{\eta}}Z_{DWF}[\eta, \bar{\eta}]|_{\eta=\bar{\eta}=0}
\end{equation}
where
\begin{align}
    Z_{DWF}[\eta, \bar{\eta}] = \int &\mathcal{D} \bar{\Psi} \mathcal{D}\Psi \mathcal{D}U \mathcal{D}\bar{\Phi}\mathcal{D}\Phi \exp{[-S_{DWF}[\bar{\Psi},\Psi,U] - S_{PF}[\bar{\Phi},\Phi,U] - S_G[U]]} \nonumber \\
    \times &\exp\left[\sum_n \bar{\eta}(n) \{ P_L \Psi(n,0) + P_R \Psi(n,N_5 - 1)\} + \sum_n \{ \bar{\Psi}(n, N_5 - 1) P_L + \bar{\Psi}(n,0)P_R\}\eta(n)\right]
\end{align}
has the sources coupling to the physical chiral field combinations

\begin{itemize}
    \item DW fermions are approximately chirally symmetric for $m = 0$. Due to the overlap of the RH and LH modes localised at the boundaries for $N_5 \neq \infty$, there is a ``residual mass'' that is generated which vanishes exponentially fast with $N_5 \rightarrow \infty$.
    \item DWF work because there is no chiral symmetry in $D = $odd dimensional spacetimes [there is no ``$\gamma_5$'' additional object in the Clifford algebra]
\end{itemize}
